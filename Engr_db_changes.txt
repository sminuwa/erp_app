
CREATE TABLE `budget_expenditures` (
  `id` bigint(20) NOT NULL AUTO_INCREMENT,
  `branch_id` bigint(20) DEFAULT NULL,
  `general_account_id` bigint(20) DEFAULT NULL,
  `budget_year` year(4) NOT NULL,
  `proposed_budget` decimal(10,2) DEFAULT NULL,
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `branch_id` (`branch_id`),
  KEY `category_id` (`general_account_id`)
) ENGINE=MyISAM AUTO_INCREMENT=1239 DEFAULT CHARSET=latin1;
